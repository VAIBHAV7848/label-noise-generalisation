\section{Detailed Mathematical Proofs}

\subsection{Lemma: Matrix Inversion Perturbation via Neumann Series}
\begin{lemma}
Let $T, \hat{T} \in \mathbb{R}^{K \times K}$ be non-singular matrices, and let $E = \hat{T} - T$ denote the estimation error with $\|E\|_2 \le \|E\|_F \le \epsilon < \frac{1}{\|T^{-1}\|_2}$. Then:
\begin{equation}
\|\hat{T}^{-1} - T^{-1}\|_2 \le \frac{\|T^{-1}\|_2^2 \epsilon}{1 - \|T^{-1}\|_2 \epsilon}.
\end{equation}
\end{lemma}
\begin{proof}
We write $\hat{T} = T + E = T (I + T^{-1} E)$. Since $\|T^{-1} E\|_2 \le \|T^{-1}\|_2 \|E\|_2 \le \|T^{-1}\|_2 \epsilon < 1$, the matrix $I + T^{-1} E$ is invertible by the Neumann series theorem, with:
\begin{equation}
(I + T^{-1} E)^{-1} = \sum_{k=0}^\infty (-T^{-1} E)^k = I - T^{-1} E + (T^{-1} E)^2 - \dots
\end{equation}
Consequently:
\begin{equation}
\hat{T}^{-1} = (I + T^{-1} E)^{-1} T^{-1} = \sum_{k=0}^\infty (-T^{-1} E)^k T^{-1}.
\end{equation}
Subtracting $T^{-1}$ yields:
\begin{equation}
\hat{T}^{-1} - T^{-1} = \sum_{k=1}^\infty (-T^{-1} E)^k T^{-1} = - (I + T^{-1} E)^{-1} T^{-1} E T^{-1}.
\end{equation}
Taking operator norms:
\begin{equation}
\|\hat{T}^{-1} - T^{-1}\|_2 \le \|(I + T^{-1} E)^{-1}\|_2 \|T^{-1}\|_2^2 \|E\|_2 \le \frac{\|T^{-1}\|_2^2 \|E\|_2}{1 - \|T^{-1} E\|_2} \le \frac{\|T^{-1}\|_2^2 \epsilon}{1 - \|T^{-1}\|_2 \epsilon}.
\end{equation}
\end{proof}

\subsection{Proposition 2A: Pointwise Expected Risk Bias}
\begin{proposition}
Let $T$ be an invertible row-stochastic transition matrix, and let $\hat{T}$ satisfy $\|E\|_F \le \epsilon < 1/\|T^{-1}\|_2$. Assume $0 \le \ell \le M$. Then for any measurable $f: \mathcal{X} \to \Delta^{K-1}$:
\begin{equation}
\left| \mathbb{E}_{\tilde{\mathcal{D}}} [\hat{T}^{-1} \vec{\ell}(f(X))]_{\tilde{Y}} - R_{\mathcal{D}}(f) \right| \le \frac{\sqrt{K} M \|T^{-1}\|_2^2 \epsilon}{1 - \|T^{-1}\|_2 \epsilon}.
\end{equation}
\end{proposition}
\begin{proof}
Let $\tilde{\mathbf{p}}(x) \in \Delta^{K-1}$ be the corrupted class posterior row vector at $x$. By total probability, $\tilde{\mathbf{p}}(x) = \mathbf{p}(x) T$. By Proposition 1, the clean risk satisfies $R_{\mathcal{D}}(f) = \mathbb{E}_X [\tilde{\mathbf{p}}(X) T^{-1} \vec{\ell}(f(X))]$.
Under imperfect estimator $\hat{T}$:
\begin{equation}
\mathbb{E}_{\tilde{\mathcal{D}}} [\hat{T}^{-1} \vec{\ell}(f(X))]_{\tilde{Y}} = \mathbb{E}_X [\tilde{\mathbf{p}}(X) \hat{T}^{-1} \vec{\ell}(f(X))].
\end{equation}
Subtracting clean risk:
\begin{align}
\left| \mathbb{E}_{\tilde{\mathcal{D}}} [\hat{T}^{-1} \vec{\ell}(f(X))]_{\tilde{Y}} - R_{\mathcal{D}}(f) \right| &= \left| \mathbb{E}_X \left[ \tilde{\mathbf{p}}(X) (\hat{T}^{-1} - T^{-1}) \vec{\ell}(f(X)) \right] \right| \\
&\le \mathbb{E}_X \left[ \|\tilde{\mathbf{p}}(X)\|_2 \|\hat{T}^{-1} - T^{-1}\|_2 \|\vec{\ell}(f(X))\|_2 \right].
\end{align}
Since $\tilde{\mathbf{p}}(X) \in \Delta^{K-1}$, $\|\tilde{\mathbf{p}}(X)\|_2 \le \|\tilde{\mathbf{p}}(X)\|_1 = 1$. Since each entry of $\vec{\ell}$ is bounded by $M$, $\|\vec{\ell}(f(X))\|_2 \le \sqrt{K} M$.
Applying Lemma 1:
\begin{equation}
\left| \mathbb{E}_{\tilde{\mathcal{D}}} [\hat{T}^{-1} \vec{\ell}(f(X))]_{\tilde{Y}} - R_{\mathcal{D}}(f) \right| \le 1 \cdot \frac{\|T^{-1}\|_2^2 \epsilon}{1 - \|T^{-1}\|_2 \epsilon} \cdot \sqrt{K} M = \frac{\sqrt{K} M \|T^{-1}\|_2^2 \epsilon}{1 - \|T^{-1}\|_2 \epsilon}.
\end{equation}
\end{proof}

\subsection{Proposition 2B: Finite-Sample Clean Excess Risk Bound}
\begin{proposition}
Assume $\ell$ is $M$-bounded and $L_{\ell, 2}$-Lipschitz with respect to the $\ell_2$ norm. Let $\hat{T}$ be a fixed matrix or computed from an independent split $S_T \perp S_R$ such that $\|\hat{T} - T\|_F \le \epsilon < 1/\|T^{-1}\|_2$. Let $\hat{f} = \arg\min_{f \in \mathcal{F}} \hat{R}_{S_R, \hat{T}}(f)$ on independent sample $S_R$ of size $n_R$. Then with probability at least $1 - \delta$:
\begin{equation}
R_{\mathcal{D}}(\hat{f}) - \min_{f \in \mathcal{F}} R_{\mathcal{D}}(f) \le \frac{2 \sqrt{K} M \|T^{-1}\|_2^2 \epsilon}{1 - \|T^{-1}\|_2 \epsilon} + 4 \sqrt{2} L_{\ell, 2} \|\hat{T}^{-1}\|_2 \mathcal{R}_{n_R}(\mathcal{F}) + 2 \sqrt{K} M \|\hat{T}^{-1}\|_2 \sqrt{\frac{\ln(2/\delta)}{2 n_R}}.
\end{equation}
\end{proposition}
\begin{proof}
Let $f^* = \arg\min_{f \in \mathcal{F}} R_{\mathcal{D}}(f)$. By standard algebraic risk decomposition:
\begin{align}
R_{\mathcal{D}}(\hat{f}) - R_{\mathcal{D}}(f^*) &= \left( R_{\mathcal{D}}(\hat{f}) - R_{\tilde{\mathcal{D}}, \hat{T}}(\hat{f}) \right) + \left( R_{\tilde{\mathcal{D}}, \hat{T}}(\hat{f}) - \hat{R}_{S_R, \hat{T}}(\hat{f}) \right) \\
&\quad + \left( \hat{R}_{S_R, \hat{T}}(\hat{f}) - \hat{R}_{S_R, \hat{T}}(f^*) \right) + \left( \hat{R}_{S_R, \hat{T}}(f^*) - R_{\tilde{\mathcal{D}}, \hat{T}}(f^*) \right) \\
&\quad + \left( R_{\tilde{\mathcal{D}}, \hat{T}}(f^*) - R_{\mathcal{D}}(f^*) \right).
\end{align}
By empirical risk minimisation optimality of $\hat{f}$, $\hat{R}_{S_R, \hat{T}}(\hat{f}) \le \hat{R}_{S_R, \hat{T}}(f^*)$, so the third term is $\le 0$.
By Proposition 2A, the first and fifth terms satisfy:
\begin{equation}
|R_{\mathcal{D}}(\hat{f}) - R_{\tilde{\mathcal{D}}, \hat{T}}(\hat{f})| \le \frac{\sqrt{K} M \|T^{-1}\|_2^2 \epsilon}{1 - \|T^{-1}\|_2 \epsilon}, \quad |R_{\tilde{\mathcal{D}}, \hat{T}}(f^*) - R_{\mathcal{D}}(f^*)| \le \frac{\sqrt{K} M \|T^{-1}\|_2^2 \epsilon}{1 - \|T^{-1}\|_2 \epsilon}.
\end{equation}
The remaining terms are bounded by the uniform generalization deviation:
\begin{equation}
R_{\mathcal{D}}(\hat{f}) - R_{\mathcal{D}}(f^*) \le \frac{2 \sqrt{K} M \|T^{-1}\|_2^2 \epsilon}{1 - \|T^{-1}\|_2 \epsilon} + 2 \sup_{f \in \mathcal{F}} \left| \hat{R}_{S_R, \hat{T}}(f) - R_{\tilde{\mathcal{D}}, \hat{T}}(f) \right|.
\end{equation}
Define the function class $\mathcal{G} = \{g_f(x, \tilde{y}) = [\hat{T}^{-1} \vec{\ell}(f(x))]_{\tilde{y}} : f \in \mathcal{F}\}$. For each sample $(x_i, \tilde{y}_i)$, $|g_f(x_i, \tilde{y}_i)| \le \|\hat{T}^{-1}\|_2 \|\vec{\ell}(f(x_i))\|_2 \le \sqrt{K} M \|\hat{T}^{-1}\|_2$.
Applying McDiarmid's inequality to the random variable $\Phi(S_R) = \sup_{f \in \mathcal{F}} |\hat{R}_{S_R, \hat{T}}(f) - R_{\tilde{\mathcal{D}}, \hat{T}}(f)|$ with bounded differences $c_i = \frac{2 \sqrt{K} M \|\hat{T}^{-1}\|_2}{n_R}$, with probability at least $1 - \delta$:
\begin{equation}
\sup_{f \in \mathcal{F}} \left| \hat{R}_{S_R, \hat{T}}(f) - R_{\tilde{\mathcal{D}}, \hat{T}}(f) \right| \le 2 \mathcal{R}_{n_R}(\mathcal{G}) + \sqrt{K} M \|\hat{T}^{-1}\|_2 \sqrt{\frac{\ln(2/\delta)}{2 n_R}}.
\end{equation}
By Maurer's vector-valued Rademacher contraction inequality for vector Lipschitz functions:
\begin{equation}
\mathcal{R}_{n_R}(\mathcal{G}) \le \sqrt{2} L_{\ell, 2} \|\hat{T}^{-1}\|_2 \mathcal{R}_{n_R}(\mathcal{F}).
\end{equation}
Combining terms gives the stated bound.
\end{proof}
